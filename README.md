# install
```bash
[ -d $HOME/.config/rofi ] && mv $HOME/.config/rofi{,.bak}
git clone https://github.com/ilgizar-khis/rofi.git $HOME/.config/rofi
```
or

```bash
[ -d $HOME/.config/rofi ] && mv $HOME/.config/rofi{,.bak}
git clone https://github.com/ilgizar-khis/rofi.git
ln -s $PWD/rofi $HOME/.config/rofi
```
